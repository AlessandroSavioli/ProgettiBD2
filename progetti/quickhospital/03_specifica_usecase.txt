Use-Case Gestione ricoveri

	nuovo_paziente(
		nom:Stringa,
		cogn:Stringa,
		nasc:Data,
		insnum:Telefono[1..*],
		ema:Email,
		post:Indirizzo
	): Paziente
		pre:
			!EXISTS p' |
				email(p', ema)
		post:
			il livello estensionale finale differisce
			da quello iniziale come segue:

			nuovi elementi del dominio: alpha
			elementi rimossi dal dominio: nessuno
			nuove ennuple:
				- Persona(alpha)
				- nome(alpha, nom)
				- cognome(alpha, cogn)
				- nascita(alpha, nasc)
				- Paziente(alpha)
				- num_telefono(alpha, tel) per ogni tel in insnum
				- email(alpha, ema)
				- indirizzo_postale(alpha, post)
			ennuple rimosse: nessuna
			valore di ritorno: result = alpha
			
	nuovo_ricovero(
		p:Paziente,
		m:Medico,
		l:Letto,
		d:Data
	): Ricovero
		pre:
			is_occupato(l, false)
			and
			d >= oggi
			and
			!EXISTS ric' |
				paz_ric(p, ric') and
				!RicoveroTerminato(ric')
		post:
			il livello estensionale finale differisce 
			da quello iniziale come segue:

			nuovi elementi del dominio: alpha
			elementi rimossi dal dominio: nessuno
			nuove ennuple:
				- Ricovero(alpha)
				- data(alpha, d)
				- paz_ric(p, alpha)
				- med_ric(m, alpha)
				- let_ric(l, alpha)
			ennuple rimosse: nessuna
			valore di ritorno: result = alpha

	termina_ricovero(r:Ricovero): RicoveroTerminato
		pre:
			!RicoveroTerminato(r)
		post:
			il livello estensionale finale differisce
			da quello iniziale come segue:

			nuovi elementi del dominio: nessuno
			elementi rimossi dal dominio: nessuno
			nuove ennuple:
				- RicoveroTerminato(r)
				- dimissione(r, oggi)
			ennuple rimosse: nessuna
			valore di ritorno: result = r

	trova_medici(pres:PrestazioneEsterna): Medico[0..*]
		pre:
			!EXISTS med |
				med_pres(med, pres)
		post:
			lo use-case non modifica il livello estensionale

			result soddisfa la seguente formula

			Mprim = { m | EXISTS spec |
							pre_spe(pres, spec) and
							primaria(m, spec) }
			and
			Msec = { m | EXISTS spec |
							pre_spe(pres, spec) and
							secondaria(m, spec) }
			and
			( |Mprim| > 0 ) -> result = Mprim
			and
			( |Mprim| = 0 and |Msec| > 0 ) -> result = Msec
			and
			( |Mprim| = 0 and |Msec| = 0 ) -> result = null
