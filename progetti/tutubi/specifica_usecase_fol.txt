	cerca_video(
		cat:Categoria,
		instag:Tag[0..*],
		v:Intero 0..5 [0..1]
	): Video[0..*]
	pre:
		instag valorizzato <-> v valorizzato
	post:
		questo use-case non modifica il livello estensionale

		instag non valorizzato and v non valorizzato ->
			VIDCATRISP = { vid | !VideoCensurato(vid) and
									cat_vid(cat, vid) and
									EXISTS risp |
										conta_risposte(vid, risp) and
										FORALL vid2, risp2 |
											 ( !VideoCensurato(vid2) and
											 cat_vid(cat, vid2) and
											 conta_risposte(vid2, risp2) )
											 -> risp >= risp2 }
			and
			result = VIDCATRISP
		and
		instag valorizzato and v valorizzato ->
			VIDCATTAG = { vid | !VideoCensurato(vid) and
								cat_vid(cat, vid) and
								EXISTS t |
									t in instag and
									tag_vid(t, vid) }
			and
			VIDVAL = { vid | vid in VIDCATTAG and
								( ( EXISTS val |
									valutazione_media(vid, val) and
									val >= v )
								or
									ha_valutazioni(vid, false) ) }
			and
			result = VIDVAL
