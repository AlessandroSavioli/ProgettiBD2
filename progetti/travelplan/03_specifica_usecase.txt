	trova_città_popolari(ini:DataOra, fin:DataOra): Città[0..*]
		pre:
			nessuna
		post:
			questo use-case non modifica il livello estensionale
			result soddisfa la seguente formula:

			POPOLARITÀ = { (c, n) | Città(c) AND
					n = |{ via | EXISTS att, luo, iniv, finv |
							att_via(att, via) AND
							inizio(via, iniv) AND
							fine(via, finv) AND
							iniv >= ini AND
							finv <= fin AND
							( att_luo(att, luo) 
							 OR arrivo(att, luo) ) AND
							citt_luo(c, luo) AND}| }
			AND
			maxn = max( n | EXISTS c | (c, n) in POPOLARITÀ )
			AND
			result = { c | EXISTS num | 
							(c, num) in POPOLARITÀ AND
							num = maxn }
