		cit:Città,
		i:Data,
		f:Data,
		posti:Intero>0
	): Utente[0..*]
		pre:
			i < f
		post:
			questo use-case non modifica il livello estensionale

			result = { u | EXISTS abi |
							citt_ute(cit, u) AND

	inoltra_richiesta(
		abi:Abitazione,
		i:Data,

		posti:PostoLetto[1..*]
	): Richiesta
