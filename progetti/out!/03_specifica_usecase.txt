	scegli_posti(evento:Evento, biglietti:Intero > 0): Posto[1..*]
		pre: 
			Postiliberi = { p | EXISTS set, sal |
									sett_pos(set, p) AND
									sala_sett(sal, set) AND
									eve_sala(evento, sal) AND
									!EXISTS b, tar |
										bigl_tarif(b, tar) AND
										eve_tarif(evento, tar) AND
										bigl_pos(b, p) }
			AND 
			|Postiliberi| >= biglietti
		post:
			questo use-case non modifica il livello estensionale

			result soddisfa la seguente formula:

			result = Postiliberi[1:biglietti]

- Use-Case Strumenti prenotazione

	nuova_prenotazione(
		biglietti: Intero > 0,
		tariffe: Tariffa[1..*],
		evento: Evento
	)
		pre:
			EXISTS inizio |
				inizio_spettacolo(evento, inizio)
			AND
			inizio > adesso
			AND
			biglietti = |tariffe|
		post:
			il modello estensionale finale differisce
			da quello iniziale come segue:

			nuovi elementi del dominio: 
				- alpha (prenotazione)
				- un insieme di elementi B = {beta_1, ..., beta_n}
				  con n = biglietti
			elementi rimossi dal dominio: nessuno
			nuove ennuple:
				- Prenotazione(alpha)
				Sia ids = { ide | EXISTS pren |
									id(pren, ide) }
				- id(alpha, MAX(id in ids) + 1)
				- eve_pren(evento, alpha)
				
				sia u l'istanza dell'utente che ha invocato lo use-case
				- clie_pren(u, alpha)

				// (Selezione dei posti)
                Sia P_scelti l'esito dello use-case scegli_posti(evento, biglietti)
                Siano p_1, p_2, ..., p_n gli elementi di P_scelti.
                Siano t_1, t_2, ..., t_n gli elementi della lista 'tariffe'.
                
				per ogni i in range(1, biglietti)
				- Biglietto(beta_i)
				- emissione(beta_i, adesso)
				- bigl_pren(beta_i, alpha)
				- bigl_pos(beta_i, p_i)
				- bigl_tarif(beta_i, t_i)
