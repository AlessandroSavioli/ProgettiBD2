- Use-Case Pre-registrazione

	pre_registrazione_priv(
		cf:CF,
		nom:Stringa,
		cog:Stringa,
		nasc:Data,
		res:Indirizzo,
		codPat:CodPat,
		scadPat:Data,
		numCart:NumCart,
		scadCart:Data
	): SocioPrivato
		
		pre:
			!EXISTS s' |
				cf(s', cf) 
			AND
			!EXISTS s', pat' |
				pat_soc(pat', s') AND
				codice(pat', codPat)
			AND
			!EXISTS s', cart' |
				cart_soc(cart', s') AND
				numero(cart', numCart)
		post:
			il livello estensionale finale 
			differisce da quello iniziale 
			come segue:

			nuovi elementi del dominio: soc, pat, cart
			elementi rimossi dal dominio: nessuno
			nuove ennuple:
				- Socio(soc) 
				- cf(soc, cf)
				- nome(soc, nom)
				- cognome(soc, cog)
				- nascita(soc, nasc)
				- SocioPrivato(soc)
				- residenza(soc, res)
				- Patente(pat)
				- codice(pat, codPat)
				- scadenza(pat, scadPat)
				- pat_soc(pat, soc)
				- CartaDiCredito(cart)
				- numero(cart, numCart)
				- scadenza(cart, scadCart)
				- cart_soc(cart, soc)
			ennuple rimosse: nessuna
			valore di ritorno: result = soc

	pre_registrazione_dip(
		cf:CF,
		nom:Stringa
		cog:Stringa
		nasc:Data

	inizia_noleggio(scard:SmartCard, aut:Auto, danni_not:Stringa[0..1]): Noleggio
		pre:
			sia s:Socio l'istanza dell'attore che
			ha invocato lo use-case

			smart_soc(scard, s)
			AND
			!EXISTS nol' |
				nol_smart(nol', scard) AND
				( !NoleggioTerminato(nol') OR
				(!NoleggioTerminato(nol') AND auto_nol(aut, nol')) )
		post:
			il livello estensionale finale
			differisce da quello iniziale
			come segue:

			nuovi elementi del dominio: alpha
			elementi rimossi dal dominio: nessuno
			nuove ennuple:
				- Noleggio(alpha)
				- inizio(alpha, adesso)
				- danni_not != null -> danni_notati(alpha, danni_not)
				- nol_smart(alpha, scard)
				- auto_nol(aut, alpha)
			ennuple rimosse: nessuna
			valore di ritorno: result = alpha

	classe_rischio(soc:Socio): 0..10
		pre:
			nessuna
		post:
			questo use-case non modifica il
			livello estensionale
			result soddisfa la seguente formula:

			EXISTS nSin, nSinNoCon |
				n_sin_no_controparte(soc, oggi - "3 anni", nSin)
				AND
				n_sin_con_controparte(soc, oggi - "3 anni", nSinNoCon)
				AND
				tot = nSin + nSinNoCon
				AND
				tot = 0 -> result = 0
				AND
				(tot > 0 AND tot < 10) -> result = tot
				AND
				tot >= 10 -> result = 10

Use-Case Strumenti addebito

	calcola_addebito(soc:Socio, ini:Data, fin:Data):Denaro
		pre:
			EXISTS scard, nol |
				smart_soc(scard, soc) AND
				nol_smart(nol, scard)
		post:
			questo use-case non modifica il
			livello estensionale
			result soddisfa la seguente formula:

			NOLEGGI = { (nol, prezz) | 
							EXISTS scard, in, fn |
								smart_soc(scard, soc) AND
								nol_smart(nol, scard) AND
								prezzo(nol, prezz) AND
								NoleggioTerminato(nol) AND
								inizio(nol, in) AND
								fine(nol, fn) AND
								ini <= in AND
								fin >= fn }
			AND
			costo_tot = SUM(prezz | EXISTS nol | 
								(nol, prezz) in NOLEGGI)
			AND
			rischio_soc = classe_rischio(soc)
			AND
			rischio_soc < 2 -> sconto_rischio = 0,9
			AND
			rischio_soc >= 5 -> 
				sconto_rischio = 1 - (rischio_soc * 10)/100
			AND
			rischio_soc >= 2 AND rischio_soc < 5 ->
				sconto_rischio = 1
			AND
			abituale(soc) -> sconto_fed = 0.85
			AND
			!abituale(soc) -> sconto_fed = 1
			AND
			result = costo_tot * sconto_rischio * sconto_fed
